    Reference model and the router module UVC¶

    Open on the project map →

    Why a reference model¶

    The scoreboard of Lab 9A expects every packet to come out. The router drops packets — when disabled, when oversized, when addressed to 3 — and the rules depend on registers written over the HBUS. Something has to know those rules and the current register values: the reference model.

    flowchart LR
    H["hbus monitor"] -->|"write_hbus: mirror every register,<br/>check every read"| R
    Y["yapp monitor"] -->|"write_yapp"| R
    R["router_reference<br/>forward if enabled,<br/>len ≤ maxpktsize, addr ≠ 3;<br/>counters, memories"] -->|"yapp_valid_out"| S["router_scoreboard<br/>yapp_in"]
    R -->|"decision_out"| C["router_coverage"]
    //------------------------------------------------------------------------------
// router_reference.sv -- reference model of the router (Lab 9B, grown into a
// full model of the registers for the random tests)
//
//   HBUS monitor ──port──► hbus_in (imp) : writes update the register mirror
//                                          and the scratch memory; reads are
//                                          checked against the model
//                                          (`check_reads`, off by default)
//   YAPP monitor ──port──► yapp_in (imp) : forward the packet on
//                                          yapp_valid_out only if the router
//                                          will route it; update the counters,
//                                          mem_size_reg and yapp_pkt_mem;
//                                          publish the decision on
//                                          decision_out (router_coverage)
//
// The model follows docs/dut/spec.md ("Behaviour rules" and "Decisions"):
//   * only packets received while router_en = 1 count, set mem_size_reg and
//     are stored in yapp_pkt_mem (dropped ones too); router_en = 0 is "deaf"
//   * each counter uses its enable bit in en_reg at the end of the packet and
//     wraps at 255
//   * yapp_pkt_mem[0] = header {length, addr}, [1..length] = payload; a shorter
//     packet leaves the old tail; the memories are not reset, so a byte is
//     only checked after it has been written
//   * unmapped addresses read 0; writes to read-only locations are ignored
// The decision is taken when the monitor publishes the packet (after its
// parity byte); the DUT takes it at the header. They agree as long as the
// configuration does not change in the middle of a packet, which the spec
// leaves undefined and the tests never do.
// A backdoor poke() is not seen here, so tests that poke keep check_reads off.
//
// `_yapp` was declared by router_scoreboard.sv (same package); only `_hbus`
// is new here.
//------------------------------------------------------------------------------
`uvm_analysis_imp_decl(_hbus)

class router_reference extends uvm_component;

  // Inputs
  uvm_analysis_imp_yapp #(yapp_packet,      router_reference) m_yapp_in;
  uvm_analysis_imp_hbus #(hbus_transaction, router_reference) m_hbus_in;
  // Output: packets the router is expected to deliver
  uvm_analysis_port #(yapp_packet) m_yapp_valid_out;
  // Output: every packet received while enabled, with the decision and the
  // configuration it met (router_coverage subscribes)
  uvm_analysis_port #(router_decision) m_decision_out;

  // Register mirror (reset values of the DUT)
  bit[7:0] m_ctrl_reg = 8'h3f;
  bit[7:0] m_en_reg   = 8'h01;
  bit[5:0] m_maxpktsize = 6'h3f;             // = ctrl_reg[5:0]
  bit       m_router_en  = 1'b1;              // = en_reg[0]
  bit[7:0] m_parity_err_cnt;
  bit[7:0] m_oversized_cnt;
  bit[7:0] m_addr_cnt[4];                    // addr0..addr3 counters
  bit[7:0] m_mem_size;
  bit[7:0] m_pkt_mem[64];
  bit       m_pkt_mem_known[64];              // written since time 0
  bit[7:0] m_scratch_mem[256];               // yapp_mem
  bit       m_scratch_known[256];

  // Knob (uvm_config_db #(bit) "check_reads"): compare every HBUS read with the model
  bit m_check_reads;

  // Statistics
  int m_packets_forwarded;
  int m_dropped_disabled;
  int m_dropped_size;
  int m_dropped_addr;
  int m_reads_checked;
  int m_reads_unknown;                        // memory bytes never written: not checked
  int m_read_mismatches;

  `uvm_component_utils(router_reference)

  extern function new(string name, uvm_component parent);
  extern function void build_phase(uvm_phase phase);

  // Keep the model up to date from the bus traffic; check the reads
  extern function void write_hbus(hbus_transaction tr);

  // Same decision the RTL makes when the header arrives, then the statistics
  extern function void write_yapp(yapp_packet pkt);

  // What a read of `addr` must return; known = 0 for a memory byte never written
  extern function bit[7:0] expected_read(bit[15:0] addr, output bit known);

  extern function void report_phase(uvm_phase phase);

endclass : router_reference

//------------------------------------------------------------------------------
function void router_reference::write_hbus(hbus_transaction tr);
  bit[7:0] exp;
  bit       known;
  if (tr.m_hwr_rd == HBUS_OP_READ) begin
    if (!m_check_reads) begin
      return;
    end
    exp = expected_read(tr.m_haddr, known);
    if (!known) begin
      m_reads_unknown++;
    end else begin
      m_reads_checked++;
      if (tr.m_hdata !== exp) begin
        m_read_mismatches++;
        `uvm_error("REF_READ", $sformatf("HBUS read 0x%04h returned 0x%02h, the model expects 0x%02h",
                                         tr.m_haddr, tr.m_hdata, exp))
      end else begin
        `uvm_info(get_type_name(), $sformatf("HBUS read 0x%04h = 0x%02h as expected", tr.m_haddr, tr.m_hdata), UVM_HIGH)
      end
    end
    return;
  end
  if (tr.m_haddr >= 16'h1100 && tr.m_haddr <= 16'h11ff) begin
    m_scratch_mem[tr.m_haddr[7:0]]   = tr.m_hdata;
    m_scratch_known[tr.m_haddr[7:0]] = 1'b1;
    return;
  end
  case (tr.m_haddr)
    16'h1000: begin
      m_ctrl_reg   = tr.m_hdata;
      m_maxpktsize = tr.m_hdata[5:0];
      `uvm_info(get_type_name(), $sformatf("maxpktsize is now %0d", m_maxpktsize), UVM_LOW)
    end
    16'h1001: begin
      m_en_reg    = tr.m_hdata;
      m_router_en = tr.m_hdata[0];
      `uvm_info(get_type_name(), $sformatf("router_en is now %0d (en_reg 0x%02h)", m_router_en, m_en_reg), UVM_LOW)
    end
    default: ;   // read-only or unmapped: the write is ignored
  endcase
endfunction : write_hbus

//------------------------------------------------------------------------------
function void router_reference::write_yapp(yapp_packet pkt);
  bit             bad;
  router_reason_e reason;
  router_decision decision;
  bad = (pkt.m_parity_type == PARITY_TYPE_BAD);
  if (!m_router_en) begin
    m_dropped_disabled++;
    `uvm_info(get_type_name(),
              $sformatf("Packet dropped: router disabled (addr=%0d len=%0d)", pkt.m_addr, pkt.m_length), UVM_LOW)
    return;                                 // deaf: no count, no memory, no pulse
  end
  if (pkt.m_addr == 2'd3) begin
    reason = ROUTER_REASON_ADDR3;
    m_dropped_addr++;
    `uvm_info(get_type_name(), $sformatf("Packet dropped: illegal address 3 (len=%0d)", pkt.m_length), UVM_LOW)
  end else if (pkt.m_length > m_maxpktsize) begin
    reason = ROUTER_REASON_OVERSIZED;
    m_dropped_size++;
    `uvm_info(get_type_name(),
              $sformatf("Packet dropped: length %0d > maxpktsize %0d (addr=%0d)", pkt.m_length, m_maxpktsize, pkt.m_addr),
              UVM_LOW)
  end else begin
    reason = ROUTER_REASON_FORWARDED;
    m_packets_forwarded++;
    `uvm_info(get_type_name(),
              $sformatf("Packet forwarded: addr=%0d len=%0d -> channel %0d", pkt.m_addr, pkt.m_length, pkt.m_addr),
              UVM_MEDIUM)
    m_yapp_valid_out.write(pkt);
  end

  // Statistics of a packet received while enabled (forwarded or dropped)
  if (bad && m_en_reg[1]) begin
    m_parity_err_cnt++;
  end
  if (pkt.m_length > m_maxpktsize && m_en_reg[2]) begin
    m_oversized_cnt++;
  end
  if (m_en_reg[4 + pkt.m_addr]) begin
    m_addr_cnt[pkt.m_addr]++;
  end
  m_mem_size = {2'b00, pkt.m_length};
  m_pkt_mem[0]       = {pkt.m_length, pkt.m_addr};
  m_pkt_mem_known[0] = 1'b1;
  for (int i = 0; i < pkt.m_length; i++) begin
    m_pkt_mem[i + 1]       = pkt.m_payload[i];
    m_pkt_mem_known[i + 1] = 1'b1;
  end

  decision            = router_decision::type_id::create("decision");
  decision.m_pkt        = pkt;
  decision.m_maxpktsize = m_maxpktsize;
  decision.m_en_reg     = m_en_reg;
  decision.m_reason     = reason;
  m_decision_out.write(decision);
endfunction : write_yapp

//------------------------------------------------------------------------------
function bit[7:0] router_reference::expected_read(bit[15:0] addr, output bit known);
  known = 1'b1;
  if (addr >= 16'h1100 && addr <= 16'h11ff) begin
    known = m_scratch_known[addr[7:0]];
    return m_scratch_mem[addr[7:0]];
  end
  if (addr >= 16'h1010 && addr <= 16'h104f) begin
    known = m_pkt_mem_known[addr - 16'h1010];
    return m_pkt_mem[addr - 16'h1010];
  end
  case (addr)
    16'h1000: return m_ctrl_reg;
    16'h1001: return m_en_reg;
    16'h1004: return m_parity_err_cnt;
    16'h1005: return m_oversized_cnt;
    16'h1006: return m_addr_cnt[3];
    16'h1009: return m_addr_cnt[0];
    16'h100a: return m_addr_cnt[1];
    16'h100b: return m_addr_cnt[2];
    16'h100d: return m_mem_size;
    default:  return 8'h00;                 // not mapped
  endcase
endfunction : expected_read

//------------------------------------------------------------------------------
function void router_reference::report_phase(uvm_phase phase);
  `uvm_info(get_type_name(), $sformatf({"\n--- Reference model report ---\n",
    "  forwarded to scoreboard   : %0d\n",
    "  dropped (router disabled) : %0d\n",
    "  dropped (oversized)       : %0d\n",
    "  dropped (address 3)       : %0d\n",
    "  counters parity/oversized/addr0..3 : %0d / %0d / %0d %0d %0d %0d"},
    m_packets_forwarded, m_dropped_disabled, m_dropped_size, m_dropped_addr,
    m_parity_err_cnt, m_oversized_cnt, m_addr_cnt[0], m_addr_cnt[1], m_addr_cnt[2], m_addr_cnt[3]), UVM_LOW)
  if (m_check_reads) begin
    `uvm_info(get_type_name(), $sformatf("HBUS reads checked against the model: %0d (%0d mismatches, %0d of memory bytes never written)",
                                         m_reads_checked, m_read_mismatches, m_reads_unknown), UVM_LOW)
  end
endfunction : report_phase

    In the course the model is tiny: it mirrors two register fields from the bus traffic and re-implements the one decision the RTL makes when a header arrives (Lab 9B). The project's version grew with the test plan and the random tests: it keeps every register, the six counters with their enables and wrap-around, mem_size, the packet memory and yapp_mem, and, when check_reads is set, compares every HBUS read with what the model expects (REF_READ errors otherwise). The routing decision is the same one; the model must agree with the DUT on the corner cases listed in the specification.

    Interface UVC vs module UVC¶

    Interface UVC (yapp, hbus, channel, clock) Module UVC (router)
    talks to pins, through an interface transactions, through TLM
    contains driver, sequencer, monitor reference model, scoreboard, coverage
    generates stimulus? yes no
    reusable across any design with that interface any testbench of this design

    router_module_env is the module UVC: an uvm_env that builds the reference, the scoreboard and the five coverage collectors of the router, connects them, and (Lab 9C) exposes exports so the testbench never reaches inside. The reference also publishes every packet it received while enabled, with its decision, on decision_out; the router_coverage collector subscribes (Functional coverage).

    //------------------------------------------------------------------------------
// router_module_env.sv -- the router MODULE UVC with exports (Lab 9C)
//
// Every external connection point is now an analysis EXPORT on the env, so
// the testbench never has to know where the imps live inside the module UVC:
//
//   yapp_export ──► reference.yapp_in       chan0_export ──► scoreboard.chan0_in
//   hbus_export ──► reference.hbus_in       chan1_export ──► scoreboard.chan1_in
//                                           chan2_export ──► scoreboard.chan2_in
//   reference.yapp_valid_out ──► scoreboard.yapp_in   (internal, unchanged)
//   reference.decision_out   ──► coverage             (router_coverage)
//   hbus_export              ──► hbus_addr_cov, hbus_bits_cov, hbus_data_cov,

//------------------------------------------------------------------------------

    The analysis-FIFO variant (Lab 9D)¶

    The same checking written as a process instead of callbacks: router_fifo_scoreboard pulls packets from uvm_tlm_analysis_fifos with blocking get() calls, keeps the register mirror in a parallel thread and checks in check_phase that every FIFO is empty.

    //------------------------------------------------------------------------------
// router_fifo_scoreboard.sv -- scoreboard + reference model built on TLM
// analysis FIFOs and blocking get() calls (Lab 9D)
//
// Instead of reacting inside write() callbacks, the checking runs as a
// PROCESS in run_phase:
//
//   yapp_export ──► yapp_fifo ──get──► [drop?] ──► chan_fifo[addr] ──get──► compare
//   hbus_export ──► hbus_fifo ──get──► mirror maxpktsize / router_en   (parallel thread)
//
// Analysis FIFOs do NOT clone: every monitor must create a new object per
// transaction (ours do), otherwise the FIFO contents get overwritten.
//------------------------------------------------------------------------------
class router_fifo_scoreboard extends uvm_scoreboard;

  // External connection points
  uvm_analysis_export #(yapp_packet)      m_yapp_export;
  uvm_analysis_export #(hbus_transaction) m_hbus_export;
  uvm_analysis_export #(channel_packet)   m_chan_export[3];

  // Storage
  uvm_tlm_analysis_fifo #(yapp_packet)      m_yapp_fifo;
  uvm_tlm_analysis_fifo #(hbus_transaction) m_hbus_fifo;
  uvm_tlm_analysis_fifo #(channel_packet)   m_chan_fifo[3];

  // Blocking readers
  uvm_get_port #(yapp_packet)      m_yapp_get;
  uvm_get_port #(hbus_transaction) m_hbus_get;
  uvm_get_port #(channel_packet)   m_chan_get[3];

  // Register mirror (DUT reset values)
  bit[5:0] m_maxpktsize = 6'h3f;
  bit       m_router_en  = 1'b1;

  // Statistics
  int m_packets_in, m_packets_matched, m_packets_mismatched;
  int m_dropped_disabled, m_dropped_size, m_dropped_addr;

  `uvm_component_utils(router_fifo_scoreboard)

  extern function new(string name, uvm_component parent);

  `include "packet_compare.sv"

  extern function void connect_phase(uvm_phase phase);
  extern task run_phase(uvm_phase phase);

  // Parallel thread: keep the register mirror current
  extern task mirror_registers();

  // Main thread: one YAPP packet at a time, in order
  extern task check_packets();

  // Every FIFO must be empty when the test ends
  extern function void check_phase(uvm_phase phase);
  extern function void report_phase(uvm_phase phase);

endclass : router_fifo_scoreboard

// router_fifo_scoreboard -- method implementations

function router_fifo_scoreboard::new(string name, uvm_component parent);
  super.new(name, parent);
  m_yapp_export = new("yapp_export", this);
  m_hbus_export = new("hbus_export", this);
  m_yapp_fifo   = new("yapp_fifo", this);
  m_hbus_fifo   = new("hbus_fifo", this);
  m_yapp_get    = new("yapp_get", this);
  m_hbus_get    = new("hbus_get", this);
  foreach (m_chan_fifo[i]) begin
    m_chan_export[i] = new($sformatf("chan%0d_export", i), this);
    m_chan_fifo[i]   = new($sformatf("chan%0d_fifo", i), this);
    m_chan_get[i]    = new($sformatf("chan%0d_get", i), this);
  end
endfunction : new

//------------------------------------------------------------------------------
function void router_fifo_scoreboard::connect_phase(uvm_phase phase);
  // exports -> FIFO analysis exports (monitors write into the FIFOs)
  m_yapp_export.connect(m_yapp_fifo.analysis_export);
  m_hbus_export.connect(m_hbus_fifo.analysis_export);
  // get ports -> FIFO get/peek exports (we read out of the FIFOs)
  m_yapp_get.connect(m_yapp_fifo.get_peek_export);
  m_hbus_get.connect(m_hbus_fifo.get_peek_export);
  foreach (m_chan_fifo[i]) begin
    m_chan_export[i].connect(m_chan_fifo[i].analysis_export);
    m_chan_get[i].connect(m_chan_fifo[i].get_peek_export);
  end
endfunction : connect_phase

//------------------------------------------------------------------------------
task router_fifo_scoreboard::run_phase(uvm_phase phase);
  fork
    mirror_registers();
    check_packets();
  join
endtask : run_phase

//------------------------------------------------------------------------------
task router_fifo_scoreboard::mirror_registers();
  hbus_transaction tr;
  forever begin
    m_hbus_get.get(tr);
    if (tr.m_hwr_rd == HBUS_OP_WRITE) begin
      case (tr.m_haddr)
        16'h1000: m_maxpktsize = tr.m_hdata[5:0];
        16'h1001: m_router_en  = tr.m_hdata[0];
        default: ;
      endcase
    end
  end
endtask : mirror_registers

//------------------------------------------------------------------------------
task router_fifo_scoreboard::check_packets();
  yapp_packet    yp;
  channel_packet cp;
  forever begin
    m_yapp_get.get(yp);                    // blocks until the monitor delivers one
    m_packets_in++;
    if (!m_router_en) begin
      m_dropped_disabled++;
      `uvm_info(get_type_name(), "Packet dropped: router disabled", UVM_LOW)
    end else if (yp.m_addr == 2'd3) begin
      m_dropped_addr++;
      `uvm_info(get_type_name(), "Packet dropped: illegal address 3", UVM_LOW)
    end else if (yp.m_length > m_maxpktsize) begin
      m_dropped_size++;
      `uvm_info(get_type_name(), $sformatf("Packet dropped: length %0d > maxpktsize %0d",
                                           yp.m_length, m_maxpktsize), UVM_LOW)
    end else begin
      m_chan_get[yp.m_addr].get(cp);         // blocks until the channel delivers it
      if (comp_equal(yp, cp)) begin
        m_packets_matched++;
        `uvm_info(get_type_name(), $sformatf("Channel %0d packet matched", yp.m_addr), UVM_MEDIUM)
      end else begin
        m_packets_mismatched++;
        `uvm_error(get_type_name(), $sformatf("Channel %0d packet MISMATCH\nexpected:\n%s\nreceived:\n%s",
                                              yp.m_addr, yp.sprint(), cp.sprint()))
      end
    end
  end
endtask : check_packets

//------------------------------------------------------------------------------
function void router_fifo_scoreboard::check_phase(uvm_phase phase);
  if (m_yapp_fifo.used() != 0) begin
    `uvm_error(get_type_name(), $sformatf("%0d YAPP packets still unchecked", m_yapp_fifo.used()))
  end
  foreach (m_chan_fifo[i]) begin
    if (m_chan_fifo[i].used() != 0) begin
      `uvm_error(get_type_name(), $sformatf("%0d packets left in channel %0d FIFO", m_chan_fifo[i].used(), i))
    end
  end
endfunction : check_phase

//------------------------------------------------------------------------------
function void router_fifo_scoreboard::report_phase(uvm_phase phase);
  `uvm_info(get_type_name(), $sformatf({"\n--- FIFO scoreboard report ---\n",
    "  packets received  : %0d\n",
    "  packets matched   : %0d\n",
    "  packets mismatched: %0d\n",
    "  dropped (disabled/oversized/addr3): %0d / %0d / %0d"},
    m_packets_in, m_packets_matched, m_packets_mismatched,
    m_dropped_disabled, m_dropped_size, m_dropped_addr), UVM_LOW)
endfunction : report_phase
